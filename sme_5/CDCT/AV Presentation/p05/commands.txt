# Run inside the Mininet VM (VirtualBox) after importing the OVA.
# Host OS: Windows 10/11 → VirtualBox → Mininet VM (login: mininet / mininet)

# ============================================================
# 0. VM SETUP (do before recording, then fast-forward)
# ============================================================
# On Windows host:
# 1. Install VirtualBox (default options)
# 2. Download Mininet VM (OVA) from https://mininet.org/download/
# 3. VirtualBox → File → Import Appliance → select .ova → Import
# 4. Start VM → login: mininet / mininet

# ============================================================
# 1. VERIFY MININET INSTALL
# ============================================================
sudo mn --version
# Expect: 2.3.x (or similar)

# ============================================================
# 2. PREPARE LAB FOLDER & COPY SCRIPT
# ============================================================
mkdir -p ~/cdct-lab
cd ~/cdct-lab

# Option A: Shared Folder (set up in VirtualBox Settings → Shared Folders → Auto-mount)
# cp /media/sf_Shared/p05_mininet_2switch_4host.py .

# Option B: Drag-and-drop (enable in VirtualBox → Devices → Drag and Drop → Bidirectional)

# Option C: SCP from host (if SSH enabled in VM)
# scp p05_mininet_2switch_4host.py mininet@<VM_IP>:~/cdct-lab/

# Verify
ls -l p05_mininet_2switch_4host.py

# ============================================================
# 3. RUN THE TOPOLOGY SCRIPT (MAIN DEMO)
# ============================================================
sudo python3 p05_mininet_2switch_4host.py

# --- Expected output (watch for these lines) ---
# *** Creating network
# *** Adding controller
# *** Adding hosts: h1 h2 h3 h4
# *** Adding switches: s1 s2
# *** Adding links: (h1,s1) (h2,s1) (h3,s2) (h4,s2) (s1,s2)

# 3 packets transmitted, 3 received, 0% packet loss
# mininet>

# ============================================================
# 4. INTERACTIVE CLI EXPLORATION (at mininet> prompt)
# ============================================================

# List all nodes
mininet> nodes
# h1 h2 h3 h4 s1 s2

# Show full topology with IPs
mininet> net
# h1 h1-eth0:s1-eth1
# h2 h2-eth0:s1-eth2
# h3 h3-eth0:s2-eth1
# h4 h4-eth0:s2-eth2
# s1 lo:  s1-eth1:h1-eth0 s1-eth2:h2-eth0 s1-eth3:s2-eth3
# s2 lo:  s2-eth1:h3-eth0 s2-eth2:h4-eth0 s2-eth3:s1-eth3

# Manual ping from h1 to h4
mininet> h1 ping -c 3 h4
# PING 10.0.0.4 ... 3 packets transmitted, 3 received

# SHOW LEARNED FLOWS ON SWITCH s1 (KEY MOMENT)
mininet> s1 dpctl dump-flows
# Example output:
# cookie=0x0, duration=12.345s, table=0, n_packets=6, n_bytes=504, idle_age=2, priority=32768,icmp,in_port=1 actions=output:2
# cookie=0x0, duration=12.340s, table=0, n_packets=6, n_bytes=504, idle_age=2, priority=32768,icmp,in_port=2 actions=output:1
# cookie=0x0, duration=12.335s, table=0, n_packets=6, n_bytes=504, idle_age=2, priority=32768,icmp,in_port=3 actions=output:3
# ... (flows learned after first ARP/ICMP exchange)

# Show virtual links (veth pairs)
mininet> links
# h1-eth0<->s1-eth1
# h2-eth0<->s1-eth2
# h3-eth0<->s2-eth1
# h4-eth0<->s2-eth2
# s1-eth3<->s2-eth3

# Optional: open xterm on h1 (if GUI works)
# mininet> xterm h1
# (closes with exit)

# Exit CLI (cleans up network, returns to bash)
mininet> exit

# ============================================================
# 5. POST-RUN VERIFICATION
# ============================================================
# Back at bash prompt
echo "Mininet lab complete."

# ============================================================
# 6. QUICK REFERENCE — MININET CLI COMMANDS
# ============================================================
# nodes              # list all hosts/switches
# net                # show topology + connections
# links              # show virtual ethernet pairs
# h1 ifconfig        # show h1's interfaces + IPs
# h1 ping -c 3 h4    # ping from h1 to h4
# h2 traceroute h3   # trace path h2 -> s1 -> s2 -> h3
# s1 dpctl dump-flows # show OpenFlow flow table on s1
# s1 dpctl add-flow ... # manually add flow (advanced)
# xterm h1           # open terminal inside h1 (needs X11)
# exit               # quit CLI, cleanup network
# help               # list all commands
